        <!-- ============ TAX ============ -->

        <section class="section" id="tip-on-tax">

          <h2>Do you tip on tax, or before tax?</h2>

          <div class="prose">

            <p>Strictly speaking, etiquette says tip on the <strong>pre-tax subtotal</strong> — the tip pays your server, not the tax office. In practice, most people tip on the after-tax total because it's the number printed at the bottom of the check, and the difference is small: on a $60 check with 8% sales tax, a 20% tip is <b>$12.00</b> on the subtotal versus <b>$12.96</b> on the taxed total — under a dollar.</p>

            <p>Pick one approach, be consistent, and don't sweat it. If you want to be precise, enter the pre-tax amount in the <a href="#calculator">calculator above</a>.</p>

          </div>

        <!-- ============ FAQ ============ -->

        <section class="section" id="faq">

          <h2>Tipping questions, answered</h2>

          <div class="faq">

            <details>

              <summary>How much do you tip on a $50 bill?</summary>

              <p>At 18%, the tip is $9.00 for a total of $59.00. At 20%, the tip is $10.00 for a total of $60.00. Split two ways, that works out to $29.50 or $30.00 per person.</p>

            </details>

            <details>

              <summary>Do you tip for takeout?</summary>

              <p>Tipping on takeout is optional. Many people skip it or leave about 10%, since there is no table service. If the kitchen handled a large or complicated order, a few extra dollars is a nice gesture.</p>

            </details>

            <details>

              <summary>What if my bill already includes a service charge?</summary>

              <p>Many restaurants add an automatic 18–20% gratuity for parties of six or more, and some add a service fee to every check. Look at the itemized bill before tipping — if a gratuity is already there, you don't need to add the full amount again, though extra for standout service is always welcome.</p>

            </details>

            <details>

              <summary>Should you tip on the discounted price or the original?</summary>

              <p>If you used a coupon or discount, the customary move is to tip on the pre-discount amount, since your server did the same work either way. Tipping on what you actually paid is also common — just remember the difference comes out of your server's pocket.</p>

            </details>

            <details>

              <summary>Is tipping the same outside the United States?</summary>

              <p>No — norms vary a lot. In much of Europe, service is included in the price (look for “service compris”) and rounding up the bill is a pleasant gesture. In Japan, tipping can be unnecessary or even awkward. A quick search before you travel beats an awkward moment at the table.</p>

            </details>

            <details>

              <summary>How do you split a bill when everyone ordered differently?</summary>

              <p>This calculator splits everything evenly, which is fastest for groups with similar checks. For uneven orders, either split the food cost exactly and tip evenly, or have each person tip 20% on what they ordered and round their share up to the next dollar.</p>

            </details>

          </div>

    /* ---------- number tweening (skipped under reduced motion) ---------- */

    var anims = new WeakMap();

    function tweenText(el, to, fmt) {

      var st = anims.get(el) || {};

      if (st.raf) cancelAnimationFrame(st.raf);

      var from = (typeof el._v === "number") ? el._v : to;

      el._v = to;

      if (reduce.matches || from === to || !isFinite(from)) { el.textContent = fmt(to); return; }

      var t0 = performance.now(), dur = 240;

      var step = function (t) {

        var p = Math.min(1, (t - t0) / dur), e = 1 - Math.pow(1 - p, 3);

        el.textContent = fmt(from + (to - from) * e);

        if (p < 1) { st.raf = requestAnimationFrame(step); anims.set(el, st); }

      };

      st.raf = requestAnimationFrame(step);

      anims.set(el, st);

    }

    function resetText(el) {

      var st = anims.get(el);

      if (st && st.raf) cancelAnimationFrame(st.raf);

      el._v = null; el.textContent = dash;

    }


    /* "each person" writes to both the receipt and the diagram, in lockstep */

    var eachRaf = 0, eachShown = null, diagEachEl = null;

    function writeEach(v, fmt) {

      rEach.textContent = fmt.format(v);

      if (diagEachEl) diagEachEl.textContent = fmt.format(v);

    }

    function renderEach(v, round) {

      var fmt = round ? fmt0 : fmt2;

      if (eachRaf) cancelAnimationFrame(eachRaf);

      if (eachShown === null || reduce.matches || eachShown === v) { eachShown = v; writeEach(v, fmt); return; }

      var from = eachShown; eachShown = v;

      var t0 = performance.now(), dur = 240;

      var step = function (t) {

        var p = Math.min(1, (t - t0) / dur), e = 1 - Math.pow(1 - p, 3);

        writeEach(from + (v - from) * e, fmt);

        if (p < 1) eachRaf = requestAnimationFrame(step);

      };

      eachRaf = requestAnimationFrame(step);

    }

    function resetEach() {

      if (eachRaf) cancelAnimationFrame(eachRaf);

      eachShown = null; rEach.textContent = dash;

      if (diagEachEl) diagEachEl.textContent = dash;

    }


    /* ---------- tip state ---------- */

    /* Any non-empty numeric value counts — including 0% */

    function customActive() {

      return customEl.value !== "" && isFinite(parseFloat(customEl.value));

    }

    function setCustomClass() {

      customFace.classList.toggle("is-active", customActive());

    }

    /* Label is derived from the clamped pct, so display and math can never diverge */

    function getTip() {

      if (customActive()) {

        var p = Math.min(999, Math.max(0, parseFloat(customEl.value)));

        return { pct: p, label: p + "%" };

      }

      var sel = PRESETS.filter(function (r) { return r.checked; })[0];

      if (sel) return { pct: +sel.value, label: sel.value + "%" };

      return { pct: +lastPreset.value, label: lastPreset.value + "%" };

    }

    function savePct() {

      if (customActive()) { store.set("tiptable:tip", { t: "c", v: customEl.value }); return; }

      var sel = PRESETS.filter(function (r) { return r.checked; })[0];

      if (sel) store.set("tiptable:tip", { t: "p", v: sel.value });

    }


    /* ---------- announcements (debounced) ---------- */

    var aTimer = 0;

    function announce(text) {

      clearTimeout(aTimer);

      aTimer = setTimeout(function () { announceEl.textContent = text; }, 900);

    }


    /* ---------- party size: hold the last valid count while the field is transiently empty ---------- */

    var lastN = 1;

    function readN() {

      var n = parseInt(pplEl.value, 10);

      if (isFinite(n) && n >= 1) lastN = Math.min(99, n);

      return lastN;

    }


    /* ---------- round-table diagram ---------- */

    function buildDiagram(n, each, round) {

      var W = 320, H = 180, cx = 160, cy = 90, rx = 98, ry = 56;

      var seats = Math.min(n, 12);

      var r = seats <= 3 ? 15 : seats <= 4 ? 14 : seats <= 6 ? 12.5 : seats <= 8 ? 11.5 : seats <= 10 ? 10.5 : 9.5;

      var showNums = n <= 12;

      var fmt = round ? fmt0 : fmt2;

      var eachStr = each === null ? dash : fmt.format(each);

      var fs = eachStr.length > 11 ? 13 : eachStr.length > 8 ? 16 : 21;

      var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="Table diagram: ' + n +

              (n === 1 ? ' person' : ' people') +

              (each !== null ? ', each paying ' + eachStr : '') + '">';

      s += '<ellipse cx="' + cx + '" cy="' + cy + '" rx="' + rx + '" ry="' + ry + '" fill="var(--table)" stroke="#BCB09A" stroke-width="1.5"/>';

      s += '<ellipse cx="' + cx + '" cy="' + cy + '" rx="' + (rx - 10) + '" ry="' + (ry - 9) + '" fill="none" stroke="rgba(25,21,18,.09)" stroke-width="1"/>';

      for (var i = 0; i < seats; i++) {

        var a = -Math.PI / 2 + i * 2 * Math.PI / seats;

        var x = cx + Math.cos(a) * (rx + 17), y = cy + Math.sin(a) * (ry + 13);

        s += '<circle cx="' + x.toFixed(1) + '" cy="' + y.toFixed(1) + '" r="' + r + '" fill="var(--card)" stroke="#191512" stroke-width="1.5"/>';

        if (showNums) {

          s += '<text x="' + x.toFixed(1) + '" y="' + (y + 0.5).toFixed(1) + '" text-anchor="middle" dy=".35em" ' +

               'font-size="' + Math.max(9.5, r * 0.82).toFixed(1) + '" font-weight="600" fill="#191512">' + (i + 1) + '</text>';

        }

      }

      s += '<text x="' + cx + '" y="' + (cy - 8) + '" text-anchor="middle" class="d-amount" style="font-size:' + fs + 'px">' + eachStr + '</text>';

      s += '<text x="' + cx + '" y="' + (cy + 16) + '" text-anchor="middle" class="d-cap">EACH</text>';

      s += '</svg>';

      diagWrap.innerHTML = s;

      diagEachEl = diagWrap.querySelector(".d-amount");

      diagWrap.classList.toggle("is-empty", each === null);

      diagCap.textContent = n > 12 ? ("Splitting " + n + " ways \u00B7 showing 12 seats")

                          : n === 1 ? "The whole check is yours"

                          : "Splitting between " + n + " people";

    }


    /* ---------- the one recalculation ---------- */

    function recalc() {

      var raw = parseFloat(billEl.value);

      var has = billEl.value !== "" && isFinite(raw) && raw > 0;

      var tip = getTip();

      var n = readN();

      var round = roundEl.checked;


      tableFor.textContent = "TABLE FOR " + n;

      rPct.textContent = "@ " + tip.label;

      hint.hidden = has;


      if (!has) {

        resetText(rBill); resetText(rTip); resetText(rTotal); resetEach();

        rNote.hidden = true;

        buildDiagram(n, null, round);

        announce("");

        return;

      }


      var tipAmt = raw * tip.pct / 100;

      var total = raw + tipAmt;

      var eachExact = total / n;

      var each = round ? Math.ceil(eachExact - 1e-9) : eachExact;

      var extra = round ? each * n - total : 0;


      tweenText(rBill, raw, fmt2.format);

      tweenText(rTip, tipAmt, fmt2.format);

      tweenText(rTotal, total, fmt2.format);

      buildDiagram(n, each, round);

      renderEach(each, round);


      if (extra > 0.005) {

        rNote.hidden = false;

        rNote.textContent = "+" + fmt2.format(extra) + " extra from rounding up";

      } else {

        rNote.hidden = true;

      }


      announce(tip.label + " tip on " + fmt2.format(raw) + ": tip " + fmt2.format(tipAmt) +

               ", total " + fmt2.format(total) +

               (n > 1 ? ", split " + n + " ways, " : ", ") +

               (round ? fmt0.format(each) : fmt2.format(each)) + " each.");

    }


    /* ---------- bill input ---------- */

    billEl.addEventListener("input", function () {

      var v = billEl.value.replace(/[^\d.]/g, "");

      var d = v.indexOf(".");

      if (d > -1) v = v.slice(0, d + 1) + v.slice(d + 1).replace(/\./g, "").slice(0, 2);

      var ip = v.split(".")[0];

      if (ip.length > 9) v = ip.slice(0, 9) + v.slice(ip.length);

      if (v !== billEl.value) billEl.value = v;

      recalc();

    });


    /* ---------- custom percentage: sanitizes AND clamps ---------- */

    customEl.addEventListener("input", function () {

      var v = customEl.value.replace(/[^\d.]/g, "");

      var d = v.indexOf(".");

      if (d > -1) v = v.slice(0, d + 1) + v.slice(d + 1).replace(/\./g, "").slice(0, 2);

      if (v.length > 5) v = v.slice(0, 5);

      if (v !== "" && parseFloat(v) > 999) v = "999";

      if (v !== customEl.value) customEl.value = v;

      if (customActive()) {

        PRESETS.forEach(function (r) { r.checked = false; });

      } else {

        lastPreset.checked = true;

      }

      setCustomClass();

      savePct();

      recalc();

    });


    /* ---------- preset radios ---------- */

    PRESETS.forEach(function (r) {

      r.addEventListener("change", function () {

        if (!r.checked) return;

        lastPreset = r;

        customEl.value = "";

        setCustomClass();

        savePct();

        recalc();

      });

    });


    /* ---------- party size events ---------- */

    pplEl.addEventListener("input", function () {

      pplEl.value = pplEl.value.replace(/\D/g, "").slice(0, 2);

      recalc();

    });

    pplEl.addEventListener("blur", function () {

      pplEl.value = String(readN());

      recalc();

    });

    function bump(d) {

      pplEl.value = String(Math.min(99, Math.max(1, readN() + d)));

      recalc();

    }

    $("pplMinus").addEventListener("click", function () { bump(-1); });

    $("pplPlus").addEventListener("click", function () { bump(1); });


    /* ---------- round-up toggle ---------- */

    roundEl.addEventListener("change", function () {

      store.set("tiptable:round", roundEl.checked);

      recalc();

    });


    /* ---------- Enter blurs ---------- */

    $("calcForm").addEventListener("submit", function (e) {

      e.preventDefault();

      if (document.activeElement && document.activeElement.blur) document.activeElement.blur();

    });


    /* ---------- restore last session ---------- */

    (function restore() {

      var rr = store.get("tiptable:round");

      if (rr) roundEl.checked = !!rr;

      var s = store.get("tiptable:tip");

      if (!s) return;

      if (s.t === "p") {

        var r = PRESETS.filter(function (x) { return x.value === String(s.v); })[0];

        if (r) { r.checked = true; lastPreset = r; }

      } else if (s.t === "c") {

        var cv = String(s.v), c = parseFloat(cv);

        if (!isFinite(c) || c < 0) return;

        if (c > 999) cv = "999";

        customEl.value = cv;

        PRESETS.forEach(function (x) { x.checked = false; });

      }

      setCustomClass();

    })();


    recalc();

  })();
